\usetikzlibrary{positioning,shapes.geometric,arrows.meta,calc}

\begin{tikzpicture}[
  font=\small,
  % Stage boxes
  stage/.style={
    rectangle, rounded corners=4pt,

    minimum width=4.6cm, minimum height=2.3cm,

  },
  % Highlighted stage (Normalisation)
  highlight/.style={
    stage, draw=orange!65!black, fill=orange!7
  },
  % Arrows between stages
  flow/.style={-{Latex[length=5pt]}, thick, black!50},
  % Key-finding annotation row
  finding/.style={
    rectangle, rounded corners=3pt,

  }
]

%% ──────────────────────────────────────────────────────────────────────
%% Row 1: Stages 1–3 (left → right)

\node[stage] (S1) {
  \textbf{1.\ GIS Dataset}\\[3pt]
  $\Ncand$ candidates

  $\Nvia$ viable sites\\
  11 criteria $\cdot$ 3 groups
};

\node[stage, right=0.55cm of S1] (S2) {
  \textbf{2.\ Weights}\\[3pt]
  AHP ($\mathrm{CR}=\CR$)\\
  Dirichlet, $\kappa=200$\\
  $\ndraws$ draws
};

\node[highlight, right=0.55cm of S2] (S3) {
  \textbf{3.\ Normalisation}\\[3pt]
  Min-max / vector / ranks\\
  $w_j^{\mathrm{eff}} = w_j A_j\,/\!\sum_k w_k A_k$\\

};

%% Arrows Row 1
\draw[flow] (S1) -- (S2);

%% ──────────────────────────────────────────────────────────────────────
%% Row 2: Stages 4–6 (right → left, under Row 1)

\node[stage, below=0.65cm of S3] (S4) {
  \textbf{4.\ Aggregation}\\[3pt]

};

\node[stage, left=0.55cm of S4] (S5) {

  $\Ntrios$ feasible trios
};

\node[stage, left=0.55cm of S5] (S6) {
